SELECT
    d.device_id,
    d.location,
    ROUND(AVG(m.temperature), 2) AS avg_temperature,
    COUNT(*)                     AS measurement_count
FROM measurements m
JOIN devices d ON d.device_id = m.device_id
GROUP BY d.device_id, d.location
ORDER BY avg_temperature DESC
LIMIT 1;

SELECT
    d.device_id,
    d.location,
    COUNT(*)            AS measurement_count,
    MAX(m.created_at)   AS last_measurement_at
FROM measurements m
JOIN devices d ON d.device_id = m.device_id
GROUP BY d.device_id, d.location
ORDER BY measurement_count DESC
LIMIT 1;

SELECT
    d.device_id,
    d.location,
    s.last_seen,
    COALESCE(s.measurement_count, 0) AS measurement_count,
    CASE
        WHEN s.last_seen >= NOW() - INTERVAL '30 seconds' THEN 'online'
        ELSE 'offline'
    END AS status
FROM devices d
LEFT JOIN (
    SELECT device_id,
           MAX(created_at) AS last_seen,
           COUNT(*)        AS measurement_count
    FROM measurements
    GROUP BY device_id
) s ON s.device_id = d.device_id
ORDER BY d.device_id;
